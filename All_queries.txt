-- point 2
-- Normalized tables: countries, diseases, infectious_cases_records
CREATE TABLE pandemic.countries (
    id INT PRIMARY KEY AUTO_INCREMENT,
    country_code VARCHAR(10) NULL,
    country_name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE pandemic.diseases (
    id INT PRIMARY KEY AUTO_INCREMENT,
    disease_name VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE pandemic.infectious_cases_records (
    id INT PRIMARY KEY AUTO_INCREMENT,
    country_id INT NOT NULL,
    year INT NOT NULL,
    disease_id INT NOT NULL,
    cases_count DOUBLE, 
    FOREIGN KEY (country_id) REFERENCES countries(id),
    FOREIGN KEY (disease_id) REFERENCES diseases(id)
);

-- Fill table counties with values from infectious_cases
INSERT INTO pandemic.countries (country_name, country_code)
SELECT 
    Entity AS country_name,
    MAX(Code) AS country_code
FROM pandemic.infectious_cases
WHERE Entity IS NOT NULL AND Entity != ''
GROUP BY Entity;

-- point 3
SELECT 
    Entity,
    Code,
    AVG(CAST(Number_rabies AS DOUBLE)) AS avg_rabies,
    MIN(CAST(Number_rabies AS DOUBLE)) AS min_rabies,
    MAX(CAST(Number_rabies AS DOUBLE)) AS max_rabies,
    SUM(CAST(Number_rabies AS DOUBLE)) AS sum_rabies
FROM pandemic.infectious_cases
WHERE Number_rabies IS NOT NULL 
  AND Number_rabies != ''
GROUP BY Entity, Code
ORDER BY avg_rabies DESC
LIMIT 10;

-- point 4
ALTER TABLE pandemic.infectious_cases
ADD COLUMN first_january_date DATE,
ADD COLUMN todays_date DATE,
ADD COLUMN year_difference INT;

UPDATE pandemic.infectious_cases
SET 
    first_january_date = MAKEDATE(CAST(Year AS UNSIGNED), 1),
    todays_date = CURDATE(),
    year_difference = TIMESTAMPDIFF(YEAR, MAKEDATE(CAST(Year AS UNSIGNED), 1), CURDATE())
WHERE Year IS NOT NULL AND Year != '';

select Year,
	   first_january_date,
	   todays_date,
       year_difference
from pandemic.infectious_cases;

-- point 5
DROP FUNCTION IF EXISTS pandemic.GetYearDifference;

DELIMITER //
CREATE FUNCTION pandemic.GetYearDifference(year_text TEXT)
RETURNS INT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE year_num INT;
    SET year_num = CAST(year_text AS UNSIGNED);
    RETURN TIMESTAMPDIFF(YEAR, MAKEDATE(year_num, 1), CURDATE());
END //
DELIMITER ;

SELECT
    Year,
    CURDATE() AS todays_date,
    pandemic.GetYearDifference(Year) AS year_difference
FROM pandemic.infectious_cases
WHERE Year IS NOT NULL AND Year != '';
